  function renderCalendar() {
    const [y, mo] = UI.calMonth.split("-").map(Number);
    const startDow = new Date(y, mo - 1, 1).getDay(), daysInMonth = new Date(y, mo, 0).getDate(), prevDays = new Date(y, mo - 1, 0).getDate(), tIso = todayISO();
    let cells = ""; DOW.forEach((d) => (cells += `<div class="cal-dow">${d[0]}</div>`));
    for (let i = 0; i < 42; i++) {
      let dayNum, cls = "cal-cell", iso, out = false;
      if (i < startDow) { dayNum = prevDays - startDow + 1 + i; out = true; const pm = mo - 1 < 1 ? 12 : mo - 1, py = mo - 1 < 1 ? y - 1 : y; iso = `${py}-${String(pm).padStart(2, "0")}-${String(dayNum).padStart(2, "0")}`; }
      else if (i >= startDow + daysInMonth) { dayNum = i - startDow - daysInMonth + 1; out = true; const nm = mo + 1 > 12 ? 1 : mo + 1, ny = mo + 1 > 12 ? y + 1 : y; iso = `${ny}-${String(nm).padStart(2, "0")}-${String(dayNum).padStart(2, "0")}`; }
      else { dayNum = i - startDow + 1; iso = `${y}-${String(mo).padStart(2, "0")}-${String(dayNum).padStart(2, "0")}`; }
      if (out) cls += " out";
      if (parseISO(iso).getDay() === 0) cls += " sun";
      if (iso === tIso) cls += " today";
      const hol = window.HOLIDAY_BY_DATE[iso]; if (hol) cls += " holiday has-tip";
      cells += `<div class="${cls}" ${hol ? `title="${esc(hol.name)}" data-action="cal-day" data-iso="${iso}"` : ""}>${dayNum}</div>`;
    }
    $("#calBody").innerHTML = cells; $("#calLabel").textContent = `${MONTHS[mo - 1]} ${y}`;
  }

  // generic donut: slices = [{id,name,short,sub,color,weight,count,action}]
  function donut(slices, centerName, centerSub) {
    const data = slices.map((s) => ({ s: s, w: Math.max(s.weight, 0) }));
    let total = data.reduce((a, d) => a + d.w, 0); if (total === 0) { data.forEach((d) => (d.w = 1)); total = data.length || 1; }
    const R = 130, r = 72, cx = 160, cy = 160; let ang = -Math.PI / 2, paths = "";
    if (!data.length) paths += `<circle cx="${cx}" cy="${cy}" r="${R}" fill="var(--glass-2)"/>`;
    data.forEach((d, i) => {
      const frac = d.w / total, a2 = ang + frac * 2 * Math.PI, large = (a2 - ang) > Math.PI ? 1 : 0;
      const x1 = cx + R * Math.cos(ang), y1 = cy + R * Math.sin(ang), x2 = cx + R * Math.cos(a2), y2 = cy + R * Math.sin(a2);
      const xi2 = cx + r * Math.cos(a2), yi2 = cy + r * Math.sin(a2), xi1 = cx + r * Math.cos(ang), yi1 = cy + r * Math.sin(ang);
      const cnt = (d.s.count !== "" && d.s.count != null) ? " — " + d.s.count + " works" : "";
      paths += `<path class="slice" data-action="${d.s.action}" data-id="${d.s.id}" d="M${x1} ${y1} A${R} ${R} 0 ${large} 1 ${x2} ${y2} L${xi2} ${yi2} A${r} ${r} 0 ${large} 0 ${xi1} ${yi1} Z" fill="${d.s.color}"><title>${esc(d.s.name)}${cnt}</title></path>`;
      const mid = (ang + a2) / 2, lr = (R + r) / 2;
      if (frac > 0.07) paths += `<text class="slice-lbl" x="${cx + lr * Math.cos(mid)}" y="${cy + lr * Math.sin(mid)}" text-anchor="middle" dy=".35em">${esc((d.s.short || d.s.name).slice(0, 12))}</text>`;
      ang = a2;
    });
    paths += `<circle class="donut-center" cx="${cx}" cy="${cy}" r="${r - 4}" data-action="org-up"><title>Up one level</title></circle>`;
    paths += `<text class="dc-name" x="${cx}" y="${cy - 4}" text-anchor="middle">${esc(centerName || "")}</text><text class="dc-sub" x="${cx}" y="${cy + 13}" text-anchor="middle">${esc(centerSub || "")}</text>`;
    return `<svg viewBox="0 0 320 320" class="donut">${paths}</svg>`;
  }

  function parseCols(text, oldCols) {
    return text.split("\n").map((s) => s.trim()).filter(Boolean).map((line, i) => {
      const parts = line.split("|"); const label = parts[0].trim(); const tp = (parts[1] || "").trim().toLowerCase();
      let type = "text", options;
      if (tp.indexOf("date") === 0) type = "date";
      else if (tp.indexOf("select") === 0) { type = "select"; options = (parts[1].split(":")[1] || "").split(",").map((o) => o.trim()).filter(Boolean); }
      const id = (oldCols && oldCols[i]) ? oldCols[i].id : slug(label) + "-" + i;
      const col = { id, label, type }; if (options) col.options = options; return col;
    });
  }

  async function boot() {
    loadUI(); isSuper = localStorage.getItem(SUPER_KEY) === "1"; // first-time users default to the "My Workspace" view (loadUI); returning users keep their last view
    applyTheme(); applySidebar(); buildClock(); renderCalendar(); applyPanels();
    bind();
    await loadTeam();
    try { MY_IP = (await (await fetch("./api/whoami", { cache: "no-store" })).json()).ip; } catch (e) { MY_IP = null; }
    currentUserId = (accessCfg().ipMap || {})[MY_IP] || null;
    const host = isLoopback(MY_IP);
    const startRole = host ? sessionStorage.getItem(START_ROLE_KEY) : null;
    if (host) {
      if (startRole && startRole !== "super" && person(startRole)) { isSuper = true; currentUserId = startRole; }   // chosen earlier this session
      else if (startRole === "super") { isSuper = true; currentUserId = null; }
      else isSuper = true;   // the PC running the .bat = host superadmin (default until they choose)
    }
    realUserId = currentUserId; realSuper = isSuper;
    viewAs = sessionStorage.getItem(VIEWAS_KEY) || null; applyViewAs();
    await migrateLegacyPersonal(); syncWorkspace();
    $("#ownerName").textContent = workspaceOwnerId() ? workspaceOwnerName() : "Guest";
    render(); setSaved("saved");
    if (host && !startRole) showStartChooser(); else showWelcome();
    setInterval(pollTeam, 12000); window.addEventListener("focus", pollTeam);  // keep all users' views in sync
    setTimeout(() => { const over = vItems().filter(wiOverdue).length; if (over) toast(`⏰ ${over} item${over > 1 ? "s" : ""} overdue in your scope`, "err"); }, 800);
  }
